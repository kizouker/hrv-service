📱 Synka från iPhone / Apple Watch

Använd REST API:et för att synka HRV-mätningar direkt från HealthKit. Alla endpoints kräver JWT-token i Authorization-headern.

1
Logga in och hämta JWT-token via POST /api/auth/login. Spara token i iOS Keychain.
2
Kolla senaste synk-timestamp — använd som anchor-datum för nästa HealthKit-query.
GET /api/sync/status
3
Skicka alla HRV-samples sedan senaste synken. Duplicerade tidsstämplar ignoreras automatiskt.
POST /api/sync/healthkit

Swift-kod (iOS)

// 1. Hämta HRV-samples från HealthKit
let type = HKQuantityType(.heartRateVariabilitySDNN)
let query = HKSampleQuery(sampleType: type,
    predicate: HKQuery.predicateForSamples(
        withStart: lastSyncDate, end: Date()),
    limit: HKObjectQueryNoLimit,
    sortDescriptors: nil) { _, samples, _ in

    // 2. Mappa till API-format
    let readings = samples?.compactMap { s -> [String:Any]? in
        guard let q = s as? HKQuantitySample else { return nil }
        return [
            "measuredAt": ISO8601DateFormatter().string(from: q.startDate),
            "sdnn": q.quantity.doubleValue(for: .init(from: "ms")),
        ]
    } ?? []

    // 3. Skicka till HRV API
    var request = URLRequest(url: URL(string: "https://din-app.railway.app/api/sync/healthkit")!)
    request.httpMethod = "POST"
    request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
    request.setValue("application/json", forHTTPHeaderField: "Content-Type")
    request.httpBody = try? JSONSerialization.data(withJSONObject: [
        "sourceName": "Apple Watch",
        "appleDeviceId": UIDevice.current.identifierForVendor?.uuidString ?? "",
        "readings": readings
    ])
    URLSession.shared.dataTask(with: request).resume()
}
healthStore.execute(query)
